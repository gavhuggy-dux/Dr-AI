<script>
function uploadFiles(input) {
    if (!input.files.length) return;
    const status = document.getElementById('upload-status');
    for (const file of input.files) {
        const formData = new FormData();
        formData.append('file', file);
        formData.append('document_type', 'upload');
        status.innerHTML = `⏳ Uploading ${file.name}...`;
        fetch('/documents/upload', {method:'POST', body:formData})
            .then(r=>r.json()).then(d=>{
                if(d.status==='uploaded') location.reload();
                else status.innerHTML = `❌ ${d.error}`;
            }).catch(e=>status.innerHTML = `❌ Upload failed`);
    }
}

function uploadCamera(input) {
    if (!input.files || !input.files[0]) return;
    const file = input.files[0];
    const reader = new FileReader();
    reader.onload = function(e) {
        fetch('/api/camera-upload', {method:'POST', body: e.target.result})
            .then(r=>r.json()).then(d=>{if(d.status==='uploaded') location.reload()});
    };
    reader.readAsArrayBuffer(file);
}

// Drag and drop
const dz = document.getElementById('dropzone');
dz.addEventListener('dragover', e => { e.preventDefault(); dz.classList.add('dragover'); });
dz.addEventListener('dragleave', () => dz.classList.remove('dragover'));
dz.addEventListener('drop', e => {
    e.preventDefault();
    dz.classList.remove('dragover');
    const files = e.dataTransfer.files;
    const input = document.getElementById('file-input');
    input.files = files;
    uploadFiles(input);
});
</script>

<script>
const chat = document.getElementById('chat');
const input = document.getElementById('user-input');
const typing = document.getElementById('typing');
const emptyState = document.getElementById('empty-state');
const cameraBtn = document.getElementById('camera-btn');
const steps = document.querySelectorAll('.step');

// Show camera button on phone
if ('mediaDevices' in navigator && 'getUserMedia' in navigator.mediaDevices) {
    cameraBtn.classList.add('active');
}

let consultationData = {};
let currentStep = 1;

function autoResize(el) {
    el.style.height = 'auto';
    el.style.height = el.scrollHeight + 'px';
}

function addMessage(text, sender) {
    emptyState?.remove();
    const div = document.createElement('div');
    div.className = `msg msg-${sender}`;
    const time = new Date().toLocaleTimeString([], {hour:'2-digit', minute:'2-digit'});
    div.innerHTML = `<div class="sender">${sender === 'doctor' ? '👩‍⚕️ Dr. AI' : '👤 You'}</div>
        <div>${text}</div>
        <div class="time">${time}</div>`;
    chat.appendChild(div);
    chat.scrollTop = chat.scrollHeight;
}

function setStep(n) {
    currentStep = n;
    steps.forEach(s => {
        s.classList.remove('active', 'done');
        if (parseInt(s.dataset.step) < n) s.classList.add('done');
        if (parseInt(s.dataset.step) === n) s.classList.add('active');
    });
}

async function sendMessage() {
    const text = input.value.trim();
    if (!text) return;
    
    addMessage(text, 'patient');
    input.value = '';
    input.style.height = 'auto';
    typing.classList.add('active');
    
    // Simulate Dr. AI response
    setTimeout(() => {
        typing.classList.remove('active');
        
        let response = '';
        if (currentStep === 1) {
            response = `Thank you for telling me about your concern. I'd like to understand more.

**Could you please tell me:**
- When did this first start?
- How severe is it on a scale of 1-10?
- Has it been getting worse, staying the same, or improving?
- Is there anything that makes it better or worse?`;
            setStep(2);
        } else if (currentStep === 2) {
            response = `Thank you. Let me ask some additional questions to build a clearer picture.

**Associated symptoms:**
- Have you noticed any other symptoms alongside this?
- Any fever, fatigue, or changes in appetite/sleep?

**Timeline:**
- How long do episodes typically last?
- How often do they occur?

**Impact:**
- Is this affecting your daily activities or work?`;
            setStep(3);
        } else if (currentStep === 3) {
            response = `Let me review your medical context.

**I need to check:**
- Do you have any existing medical conditions?
- Are you currently taking any medications or supplements?
- Have you had any recent tests or investigations?
- Is there any family history of similar conditions?
- Any allergies I should know about?`;
            setStep(4);
        } else if (currentStep === 4) {
            // Check for red flags
            const emergency = ['chest pain','breathless','stroke','bleeding','severe','unconscious','paralysis','suicidal'];
            const hasEmergency = emergency.some(e => text.toLowerCase().includes(e));
            
            if (hasEmergency) {
                response = `⚠️ **IMPORTANT — Please seek urgent medical attention.**

Based on what you've described, this could be a medical emergency. Please call emergency services (999 in the UK) or go to your nearest A&E department immediately.

Do not wait. Your safety is the priority.`;
                setStep(8);
            } else {
                response = `Good, I don't see any immediate emergency signs based on what you've described. Let me proceed with my analysis.

**Based on the information gathered so far, I'll now work through the differential diagnosis.**`;
                setStep(5);
            }
        } else if (currentStep === 5) {
            response = `## Differential Diagnosis

Based on my analysis, here are the key possibilities to consider:

**Most Likely:**
- [Condition A] — the symptom profile matches most closely
- [Condition B] — also possible given the presentation

**Requires Further Investigation:**
- [Condition C] — less likely but important to rule out

**Dangerous Conditions To Rule Out:**
- None identified at this stage, but monitoring advised

**Confidence Level: Moderate** — More information from tests would improve accuracy.`;
            setStep(6);
        } else if (currentStep === 6) {
            response = `## Recommended Investigations

To confirm the diagnosis and rule out other possibilities, I recommend:

**Blood Tests:**
- Full blood count
- Inflammatory markers (CRP, ESR)
- [Relevant specific tests]

**Monitoring:**
- Track your symptoms daily
- Note any triggers or patterns

**Referrals:**
- Consider speaking with your GP
- Specialist referral if symptoms persist

Would you like me to create a summary of this consultation for your records?`;
            setStep(7);
        } else if (currentStep === 7) {
            response = `## Clinical Reasoning

Let me explain my thinking process.

**Why [Condition A] fits:**
Your described symptoms align with the typical presentation based on onset, duration, and associated features.

**Why it may not fit:**
Some aspects of your presentation are atypical, which is why I recommend further investigation.

**What the tests will tell us:**
The recommended tests will help confirm or exclude the leading possibilities and guide treatment.

**Next Steps:**
1. Monitor your symptoms
2. Arrange the recommended tests
3. Follow up with a GP if needed
4. Come back for a follow-up consultation with the results

Your health and safety are my priority. ❤️`;
            setStep(8);
            
            // Offer to save
            setTimeout(() => {
                addMessage('**Would you like me to save this consultation to your medical record?** Reply "yes" or "save" to confirm.', 'doctor');
            }, 1000);
        } else if (currentStep === 8) {
            if (text.toLowerCase().match(/^(yes|save|sure|ok)/)) {
                response = `✅ **Consultation saved to your medical record.**

**Summary:**
- Main complaint documented
- Symptoms assessed
- Risk evaluated
- Recommendations provided

You can view this consultation anytime in your **Past Consultations** page.

**Follow-up:** I recommend checking back if your symptoms change or worsen, or once you've had the recommended tests.

Take care of yourself. 💙`;
                
                // Save to backend
                fetch('/api/consult', {
                    method: 'POST',
                    headers: {'Content-Type': 'application/json'},
                    body: JSON.stringify({
                        main_complaint: 'Auto-saved consultation',
                        summary: 'Consultation completed via Dr. AI chat interface',
                        differential: 'Awaiting full analysis',
                        most_likely: 'To be determined',
                        confidence: 'Moderate',
                        follow_up: 'Recommended if symptoms persist',
                    })
                });
            } else {
                response = `Understood. You can always come back for another consultation whenever you need.

Your health records remain private and secure. If you change your mind, you can save the consultation from the Past Consultations page.

Take care! 💙`;
            }
        }
        
        addMessage(response, 'doctor');
    }, 1500 + Math.random() * 1000);
}

function openCamera() {
    const cam = document.getElementById('camera-input');
    cam.click();
}

function uploadPhoto(input) {
    if (!input.files || !input.files[0]) return;
    
    const file = input.files[0];
    addMessage(`📷 Photo uploaded: ${file.name}`, 'patient');
    
    // Upload to server
    const formData = new FormData();
    formData.append('file', file);
    formData.append('document_type', 'scan');
    formData.append('description', 'Camera scan from consultation');
    
    fetch('/documents/upload', {method: 'POST', body: formData})
        .then(r => r.json())
        .then(data => {
            if (data.status === 'uploaded') {
                addMessage(`✅ Photo saved to your documents. I can see this is a ${file.type} image. In a full clinical setting, I would analyze this image for relevant medical findings.`, 'doctor');
            }
        })
        .catch(() => addMessage('⚠️ Could not save photo. Please try again.', 'doctor'));
}

setTimeout(() => {
    addMessage(`👩‍⚕️ Hello, I'm **Dr. AI**. I'm here to help you understand your symptoms and guide you toward the right care.

Please describe your main health concern in as much detail as you can. What's been bothering you?`, 'doctor');
    emptyState?.remove();
}, 800);
</script>
